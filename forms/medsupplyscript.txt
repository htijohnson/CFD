const EMAIL_RECIPIENT = "jjohnson@coburgfire.org";

// ==========================================
// GET REQUEST HANDLER
// ==========================================
function doGet(e) {
  var action = e ? e.parameter.action : null;
  
  if (action === "history") {
    return createJsonResponse(getInspectionHistory());
  } else if (action === "inactive") {
    return createJsonResponse(getInactiveItems());
  } else if (action === "moveItem") {
    const result = moveItemLocation(
      e.parameter.oldCabinet,
      e.parameter.oldShelf,
      e.parameter.newCabinet,
      e.parameter.newShelf,
      e.parameter.name
    );
    return createJsonResponse(result);
  }
  
  // Default: Return full active inventory
  return createJsonResponse(getInventoryData());
}

// ==========================================
// POST REQUEST HANDLER
// ==========================================
function doPost(e) {
  try {
    const contents = JSON.parse(e.postData.contents);
    const action = contents.action;
    let result = { status: "error", message: "Invalid action" };

    if (action === "submitInspection") {
      result = processInspectionSubmission(contents);
    } else if (action === "addItem") {
      result = addNewItem(contents.item);
    } else if (action === "deleteItem") {
      result = markItemInactive(contents.item);
    } else if (action === "reactivateItem") {
      result = reactivateItem(contents.item);
    } else if (action === "reorderItems") {
      result = reorderShelfItems(contents.cabinet, contents.shelf, contents.orderedNames);
    }

    return createJsonResponse(result);
  } catch (err) {
    return createJsonResponse({ status: "error", message: err.toString() });
  }
}

// ==========================================
// SUBMISSION HANDLER
// ==========================================
function processInspectionSubmission(payload) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Log to "Inspection Log" sheet
  let logSheet = ss.getSheetByName("Inspection Log");
  if (!logSheet) {
    logSheet = ss.insertSheet("Inspection Log");
    logSheet.appendRow(["Timestamp", "Inspector", "Total Items", "Low Stock Count", "Items Details JSON"]);
  } else {
    // Ensure header row exists if empty
    if (logSheet.getLastRow() === 0) {
      logSheet.appendRow(["Timestamp", "Inspector", "Total Items", "Low Stock Count", "Items Details JSON"]);
    }
  }

  const items = payload.items || [];
  let lowStockCount = 0;
  
  items.forEach(i => {
    const currentQty = parseInt(i.qty !== undefined ? i.qty : i.currentQty, 10) || 0;
    const minQty = parseInt(i.min !== undefined ? i.min : i.minQty, 10) || 0;
    if (currentQty < minQty) {
      lowStockCount++;
    }
  });

  const formattedDate = payload.timestamp 
    ? payload.timestamp 
    : Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
    
  const itemsJson = JSON.stringify(items);

  // Append entry including itemized JSON into Column E
  logSheet.appendRow([formattedDate, payload.inspector, items.length, lowStockCount, itemsJson]);

  // 2. Queue payload in ScriptProperties for the background timer
  const scriptProperties = PropertiesService.getScriptProperties();
  const submissionKey = "SUBMISSION_" + new Date().getTime();
  scriptProperties.setProperty(submissionKey, JSON.stringify({
    inspector: payload.inspector,
    timestamp: formattedDate,
    items: items,
    lowStockCount: lowStockCount
  }));

  return { status: "success", message: "Inspection logged successfully" };
}

// ==========================================
// BACKGROUND EMAIL RUNNER (Triggered every 5 min)
// ==========================================
function sendDelayedEmailReport() {
  const scriptProperties = PropertiesService.getScriptProperties();
  const keys = scriptProperties.getKeys();

  keys.forEach(key => {
    if (key.indexOf("SUBMISSION_") === 0) {
      const rawData = scriptProperties.getProperty(key);
      if (rawData) {
        const data = JSON.parse(rawData);
        
        let subject = `📋 Inspection Report Submitted by ${data.inspector}`;
        if (data.lowStockCount > 0) {
          subject = `⚠️ [LOW STOCK ALERT] Inspection Report by ${data.inspector}`;
        }

        let bodyHtml = `
          <h2>Medical Supply Inspection Report</h2>
          <p><strong>Inspector:</strong> ${data.inspector}</p>
          <p><strong>Date/Time:</strong> ${data.timestamp}</p>
          <p><strong>Total Items Checked:</strong> ${data.items.length}</p>
          <p><strong>Low/Out of Stock Items:</strong> <span style="color:${data.lowStockCount > 0 ? 'red' : 'green'}; font-weight:bold;">${data.lowStockCount}</span></p>
          <hr>
          <h3>Full Inventory Breakdown:</h3>
          <table border="1" cellpadding="6" cellspacing="0" style="border-collapse:collapse; font-family:sans-serif; width:100%;">
            <tr style="background-color:#f2f2f2;">
              <th>Cabinet</th>
              <th>Shelf</th>
              <th>Item Name</th>
              <th>Min Qty</th>
              <th>Current Qty</th>
              <th>Exp. Date</th>
            </tr>
        `;

        data.items.forEach(item => {
          const qty = parseInt(item.qty !== undefined ? item.qty : item.currentQty, 10) || 0;
          const min = parseInt(item.min !== undefined ? item.min : item.minQty, 10) || 0;
          const isLow = qty < min;
          const rowBg = isLow ? "background-color:#ffe6e6;" : "";
          
          bodyHtml += `
            <tr style="${rowBg}">
              <td>${item.cabinet || 'N/A'}</td>
              <td>${item.shelf || 'N/A'}</td>
              <td><strong>${item.name || item.itemName}</strong></td>
              <td>${min}</td>
              <td>${qty}</td>
              <td>${item.exp || item.expDate || 'N/A'}</td>
            </tr>
          `;
        });

        bodyHtml += `</table>`;

        MailApp.sendEmail({
          to: EMAIL_RECIPIENT,
          subject: subject,
          htmlBody: bodyHtml
        });

        // Delete processed submission from queue
        scriptProperties.deleteProperty(key);
      }
    }
  });
}

// INVENTORY & UTILITY FUNCTIONS
// ==========================================
function getInventoryData() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory");
  if (!sheet) return {};

  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return {};

  const inventory = {};

  for (let i = 1; i < data.length; i++) {
    const [cabinet, shelf, name, minQty, currentQty, expDate, status] = data[i];
    
    if (status === "Inactive") continue;

    if (!inventory[cabinet]) inventory[cabinet] = {};
    if (!inventory[cabinet][shelf]) inventory[cabinet][shelf] = [];

    let formattedExp = "";
    if (expDate instanceof Date) {
      formattedExp = Utilities.formatDate(expDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
    } else if (expDate) {
      formattedExp = String(expDate);
    }

    inventory[cabinet][shelf].push({
      name: name,
      minQty: minQty,
      currentQty: currentQty,
      expDate: formattedExp,
      checked: false
    });
  }

  return inventory;
}

function getInspectionHistory() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inspection Log");
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  const history = [];
  for (let i = data.length - 1; i >= 1; i--) {
    let rawItems = data[i][4]; // Items JSON string stored in Column E
    let itemsParsed = [];

    if (rawItems) {
      try {
        itemsParsed = typeof rawItems === "string" ? JSON.parse(rawItems) : rawItems;
      } catch (err) {
        itemsParsed = [];
      }
    }

    history.push({
      date: data[i][0],
      inspector: data[i][1],
      totalItems: data[i][2],
      lowStockCount: data[i][3],
      items: itemsParsed
    });
  }
  return history;
}

function getInactiveItems() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory");
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  
  const inactive = [];
  for (let i = 1; i < data.length; i++) {
    if (data[i][6] === "Inactive") {
      inactive.push({ cabinet: data[i][0], shelf: data[i][1], name: data[i][2] });
    }
  }
  return inactive;
}

function addNewItem(item) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory");
  sheet.appendRow([item.cabinet, item.shelf, item.name, item.minQty, item.currentQty, item.expDate || "", "Active"]);
  return { status: "success" };
}

function markItemInactive(item) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory");
  const data = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === item.cabinet && data[i][1] === item.shelf && data[i][2] === item.name) {
      sheet.getRange(i + 1, 7).setValue("Inactive");
      return { status: "success" };
    }
  }
  return { status: "error", message: "Item not found" };
}

function reactivateItem(item) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory");
  const data = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === item.cabinet && data[i][1] === item.shelf && data[i][2] === item.name) {
      sheet.getRange(i + 1, 7).setValue("Active");
      return { status: "success" };
    }
  }
  return { status: "error", message: "Item not found" };
}

function moveItemLocation(oldCab, oldShelf, newCab, newShelf, itemName) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Inventory");
  const data = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === oldCab && data[i][1] === oldShelf && data[i][2] === itemName) {
      sheet.getRange(i + 1, 1).setValue(newCab);
      sheet.getRange(i + 1, 2).setValue(newShelf);
      return { status: "success" };
    }
  }
  return { status: "error", message: "Item not found" };
}
